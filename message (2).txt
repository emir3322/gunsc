local function create(class, props, parent)
	local obj = Instance.new(class)
	for k, v in pairs(props) do
		obj[k] = v
	end
	obj.Parent = parent
	return obj
end

local function corner(parent, radius)
	return create("UICorner", { CornerRadius = UDim.new(0, radius) }, parent)
end

local function stroke(parent, color, thickness)
	return create("UIStroke", { Color = color, Thickness = thickness or 1 }, parent)
end

local function gradient(parent, c0, c1, rot)
	return create("UIGradient", { Color = ColorSequence.new(c0, c1), Rotation = rot or 90 }, parent)
end

local function tween(obj, props, t, style, dir)
	TweenService:Create(obj, TweenInfo.new(t or 0.15, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props):Play()
end

local orders = setmetatable({}, { __mode = "k" })
local function nextOrder(page)
	orders[page] = (orders[page] or 0) + 1
	return orders[page]
end

local function applyValues(values)
	-- Hazır ayar uygulanırken özel kilitlenmemiş daire bayrağını sıfırla (ayar içinde açıkça true değilse)
	if values.unlockedCircle == nil and controls.unlockedCircle then
		controls.unlockedCircle.set(false)
	end
	for k, v in pairs(values) do
		if controls[k] then
			controls[k].set(v)
		else
			settings[k] = v
		end
	end
end

local function register(key, renderFn, hook)
	local c = controls[key]
	if not c then
		c = { renders = {}, hooks = {} }
		c.set = function(v)
			settings[key] = v
			for _, r in ipairs(c.renders) do r() end
			for _, h in ipairs(c.hooks) do h(v) end
		end
		controls[key] = c
	end
	table.insert(c.renders, renderFn)
	if hook then table.insert(c.hooks, hook) end
	table.insert(rerenders, renderFn)
end

-- KİLİTLENMEMİŞ DAİRE SKILL YÖNLENDİRME (HOOK)
------------------------------------------------------------
do
	local ok, mt = pcall(getrawmetatable, game)
	if ok and mt and setreadonly then
		local oldIndex = mt.__index
		local oldNamecall = mt.__namecall
		setreadonly(mt, false)

		mt.__index = newcclosure(function(self, k)
			if running and settings.unlockedCircle and activeAimPos and not checkcaller() then
				if self == mouse then
					if k == "Hit" then
						return CFrame.new(activeAimPos)
					elseif k == "Target" then
						return currentTarget and currentTarget.Character and getAimPart(currentTarget.Character)
					elseif k == "UnitRay" then
						local origin = camera.CFrame.Position
						return Ray.new(origin, (activeAimPos - origin).Unit)
					end
				end
			end
			return oldIndex(self, k)
		end)

		mt.__namecall = newcclosure(function(self, ...)
			local method = getnamecallmethod()
			if running and settings.unlockedCircle and activeAimPos and not checkcaller() then
				if self == workspace and method == "Raycast" then
					local args = { ... }
					if typeof(args[1]) == "Vector3" and typeof(args[2]) == "Vector3" then
						local dist = args[2].Magnitude
						args[2] = (activeAimPos - args[1]).Unit * dist
						return oldNamecall(self, unpack(args))
					end
				end
			end
			return oldNamecall(self, ...)
		end)

		setreadonly(mt, true)
	end
end

notify = function(text)
	if not settings.notifications or settings.silentUI then return end
	toastToken += 1
	local my = toastToken
	toastLabel.Text = text
	toastBar.BackgroundColor3 = THEME.accent
	tween(toastFrame, { BackgroundTransparency = 0.05 }, 0.15)
	tween(toastBar, { BackgroundTransparency = 0 }, 0.15)
	tween(toastLabel, { TextTransparency = 0 }, 0.15)

		if my == toastToken then

			tween(toastBar, { BackgroundTransparency = 1 }, 0.3)
			tween(toastLabel, { TextTransparency = 1 }, 0.3)

	end)

local function makeSlider(page, name, key, min, max, decimals, suffix, onChange)
	local scroller = page:IsA("ScrollingFrame") and page or page:FindFirstAncestorOfClass("ScrollingFrame")
	local holder = create("Frame", {
		LayoutOrder = nextOrder(page),
		Size = UDim2.new(1, 0, 0, 44),
		BackgroundTransparency = 1,
	}, page)
	create("TextLabel", {
		Size = UDim2.new(1, -80, 0, 18),
		BackgroundTransparency = 1,
		Text = name,
		TextColor3 = THEME.text,
		Font = Enum.Font.GothamMedium,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, holder)
	local valueLabel = create("TextLabel", {
		Position = UDim2.new(1, -80, 0, 0),
		Size = UDim2.fromOffset(80, 18),
		BackgroundTransparency = 1,
		TextColor3 = THEME.accent,
		Font = Enum.Font.GothamBold,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, holder)
	local track = create("Frame", {
		Position = UDim2.fromOffset(0, 30),
		Size = UDim2.new(1, 0, 0, 6),
		BackgroundColor3 = THEME.track,
		BorderSizePixel = 0,
	}, holder)
	corner(track, 3)
	local fill = create("Frame", { Size = UDim2.new(0, 0, 1, 0), BackgroundColor3 = THEME.accent, BorderSizePixel = 0 }, track)
	corner(fill, 3)
	gradient(fill, WHITE, Color3.fromRGB(190, 190, 205), 0)
	local knob = create("Frame", {
		Size = UDim2.fromOffset(16, 16),
		AnchorPoint = Vector2.new(0.5, 0.5),
		BackgroundColor3 = WHITE,
		BorderSizePixel = 0,
	}, track)
	corner(knob, 8)
	local knobStroke = stroke(knob, THEME.accent, 2)

	local mult = 10 ^ decimals
	local function render()
		local v = settings[key]
		local a = (v - min) / (max - min)
		fill.Size = UDim2.new(a, 0, 1, 0)
		knob.Position = UDim2.new(a, 0, 0.5, 0)
		valueLabel.Text = tostring(v) .. suffix
		valueLabel.TextColor3 = THEME.accent
		fill.BackgroundColor3 = THEME.accent
		knobStroke.Color = THEME.accent
	end
	local function set(v)
		v = math.clamp(math.floor(v * mult + 0.5) / mult, min, max)
		settings[key] = v
		render()
		if onChange then onChange(v) end
	end
	controls[key] = { set = set }
	table.insert(rerenders, render)

	local dragging = false
	local function fromX(x)
		local a = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		set(min + (max - min) * a)
	end
	holder.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			if scroller then scroller.ScrollingEnabled = false end
			fromX(input.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			fromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
			if scroller then scroller.ScrollingEnabled = true end
		end
	end)
	set(settings[key])
end

local function makeSegmented(page, name, key, options, onChange)
	local holder = create("Frame", {
		LayoutOrder = nextOrder(page),
		Size = UDim2.new(1, 0, 0, 56),
		BackgroundTransparency = 1,
	}, page)
	create("TextLabel", {
		Size = UDim2.new(1, 0, 0, 18),
		BackgroundTransparency = 1,
		Text = name,
		TextColor3 = THEME.text,
		Font = Enum.Font.GothamMedium,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)
	local box = create("Frame", {
		Position = UDim2.fromOffset(0, 22),
		Size = UDim2.new(1, 0, 0, 32),
		BackgroundColor3 = THEME.bg,
		BorderSizePixel = 0,
	}, holder)
	corner(box, 9)
	stroke(box, THEME.stroke, 1)
	local buttons = {}
	local n = #options
	for i, opt in ipairs(options) do
		local btn = create("TextButton", {
			Size = UDim2.new(1 / n, -4, 1, -6),
			Position = UDim2.new((i - 1) / n, 2, 0, 3),
			BackgroundColor3 = THEME.accent,
			BackgroundTransparency = 1,
			Text = opt,
			Font = Enum.Font.GothamBold,
			TextSize = 12,
			TextColor3 = THEME.subtext,
			AutoButtonColor = false,
		}, box)
		corner(btn, 7)
		buttons[opt] = btn
		btn.MouseButton1Click:Connect(function()
			controls[key].set(opt)
		end)
	end
	local function render()
		for opt, btn in pairs(buttons) do
			local active = settings[key] == opt
			btn.BackgroundColor3 = THEME.accent
			tween(btn, { BackgroundTransparency = active and 0 or 1 })
			btn.TextColor3 = active and WHITE or THEME.subtext
		end
	end
	controls[key] = {
		set = function(v)
			settings[key] = v
			render()
			if onChange then onChange(v) end
		end,
	}
	table.insert(rerenders, render)
	render()
end

local mobileBtns = {}

local function createFloatingButton(name, text, color, defaultPos, callback)
	local b = create("TextButton", {
		Name = name,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = defaultPos,
		Size = UDim2.fromOffset(settings.mobileBtnSize, settings.mobileBtnSize),
		BackgroundColor3 = color,
		Text = text,
		TextColor3 = WHITE,
		Font = Enum.Font.GothamBold,
		TextSize = 11,
		AutoButtonColor = true,
		Visible = false,
	}, gui)
	local c = corner(b, settings.mobileBtnSize / 2)
	gradient(b, WHITE, Color3.fromRGB(185, 185, 200), 90)
	local st = stroke(b, WHITE, 1.5)
	st.Transparency = 0.7

	local dragging = false
	local dragStart, startPos, moved

	b.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			if settings.mobileEditMode then
				dragging = true
				moved = false
				dragStart = input.Position
				startPos = b.Position
			end
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			if delta.Magnitude > 4 then
				moved = true
			end
			b.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			if dragging then
				dragging = false
			end
		end
	end)

	b.MouseButton1Click:Connect(function()
		if settings.mobileEditMode or moved then return end
		callback()
	end)

	table.insert(mobileBtns, { btn = b, corner = c, stroke = st })
	return b
end

local friendCache = {}
local function isFriend(p)
	if friendCache[p.UserId] == nil then
		local ok, res = pcall(function()
			return player:IsFriendsWith(p.UserId)
		end)
		friendCache[p.UserId] = ok and res or false
	end
	return friendCache[p.UserId]
end

local visParams = RaycastParams.new()
visParams.FilterType = Enum.RaycastFilterType.Exclude

function getAimPart(char)
	local part
	if settings.aimPart == "Kafa" then
		part = char:FindFirstChild("Head")
	elseif settings.aimPart == "Gövde" then
		part = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
	end
	return part or char:FindFirstChild("HumanoidRootPart")
end

currentTarget = nil

local function evaluate(other, origin, camLook, localChar)
	if other == player then return nil end
	local char = other.Character
	if not char then return nil end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hum or not hrp or hum.Health <= 0 then return nil end
	if settings.ignoreTeam and other.Team ~= nil and other.Team == player.Team then return nil end
	if settings.ignoreFriends and isFriend(other) then return nil end

	local part = getAimPart(char)
	if not part then return nil end
	local offset = part.Position - origin
	local dist = offset.Magnitude
	if dist < 0.01 or dist > settings.maxDistance or dist < settings.minDistance then return nil end

	local angle = math.deg(math.acos(math.clamp(camLook:Dot(offset.Unit), -1, 1)))
	if settings.fovLimit < 180 and angle > settings.fovLimit / 2 then return nil end

	local screenDist = nil
	if settings.circleEnabled or settings.priority == "Merkez" then
		local v = camera:WorldToViewportPoint(part.Position)
		if v.Z <= 0 then
			if settings.circleEnabled then return nil end
			screenDist = math.huge
		else
			screenDist = (Vector2.new(v.X, v.Y) - camera.ViewportSize / 2).Magnitude
			if settings.circleEnabled and screenDist > settings.circleRadius then
				return nil
			end
		end
	end

	if settings.wallCheck then
		visParams.FilterDescendantsInstances = { localChar }
		local hit = workspace:Raycast(origin, offset, visParams)
		if hit and not hit.Instance:IsDescendantOf(char) then return nil end
	end

	local score
	if settings.priority == "Mesafe" then
		score = dist
	elseif settings.priority == "Can" then
		score = hum.Health
	elseif settings.priority == "Merkez" then
		score = screenDist or dist
	else
		score = angle
	end
	return { score = score, part = part, dist = dist, hum = hum }
end

local function findTarget(origin, camLook, localChar)
	if settings.sticky and currentTarget then
		local r = evaluate(currentTarget, origin, camLook, localChar)
		if r then return currentTarget, r end
	end
	local best, bestInfo
	for _, other in ipairs(Players:GetPlayers()) do
		local r = evaluate(other, origin, camLook, localChar)
		if r and (not bestInfo or r.score < bestInfo.score) then
			best, bestInfo = other, r
		end
	end
	return best, bestInfo
end

local avatarCache, shownAvatar = {}, nil
local function setAvatar(userId)
	if shownAvatar == userId then return end
	shownAvatar = userId
	if not userId then
		avatar.Image = ""
		avatarQ.Visible = true
		return
	end
	local cached = avatarCache[userId]
	if cached then
		avatar.Image = cached
		avatarQ.Visible = false
		return
	end
	avatar.Image = ""
	avatarQ.Visible = true
	task.spawn(function()
		local ok, img = pcall(function()
			return (Players:GetUserThumbnailAsync(userId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100))
		end)
		if ok and img then
			avatarCache[userId] = img
			if shownAvatar == userId then
				avatar.Image = img
				avatarQ.Visible = false
			end
		end
	end)
end

local function setTargetInfo(name, dist, hp, maxHp, userId)
	if name then

		hpFill.Size = UDim2.new(frac, 0, 1, 0)
		hpFill.BackgroundColor3 = Color3.fromHSV(0.33 * frac, 0.75, 0.9)

		markerLabel.Text = name

		targetText.Text = settings.circleEnabled and "Dairede kimse yok" or "Hedef yok"
		hpText.Text = ""
		hpFill.Size = UDim2.new(0, 0, 1, 0)

	end

local function approach(cur, goal, dt)
	local d = (goal - cur + math.pi) % (2 * math.pi) - math.pi
	if settings.speed >= 50 then
		return cur + d
	end
	if settings.smoothing == "Lineer" then
		local maxStep = math.rad(settings.speed * 20) * dt
		return cur + math.clamp(d, -maxStep, maxStep)
	end
	return cur + d * (1 - math.exp(-settings.speed * dt))
end

local BIND_NAME = "YakinOyuncuyaBak"
local camActive = false
local camYaw, camPitch = 0, 0
local savedFOV = nil
local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude

local function enterCamera()
	if camActive then return end
	camActive = true
	local look = camera.CFrame.LookVector
	camYaw = math.atan2(-look.X, -look.Z)
	camPitch = math.asin(math.clamp(look.Y, -1, 1))
	camera.CameraType = Enum.CameraType.Scriptable
	UserInputService:GetMouseDelta()
end

local function exitCamera()
	if not camActive then return end
	camActive = false
	camera.CameraType = Enum.CameraType.Custom
	UserInputService.MouseBehavior = Enum.MouseBehavior.Default
end

local function mouseOverMenu()
	if not main.Visible then return false end
	local m = UserInputService:GetMouseLocation()
	local p, s = main.AbsolutePosition, main.AbsoluteSize
	return m.X >= p.X and m.X <= p.X + s.X and m.Y >= p.Y and m.Y <= p.Y + s.Y
end

UserInputService.InputChanged:Connect(function(input, processed)
	if processed or not camActive then return end
	if input.UserInputType ~= Enum.UserInputType.Touch then return end
	if main.Visible then
		local p, a, s = input.Position, main.AbsolutePosition, main.AbsoluteSize
		if p.X >= a.X and p.X <= a.X + s.X and p.Y >= a.Y and p.Y <= a.Y + s.Y then
			return
		end
	end
	local sens = 0.004 * settings.sensitivity
	camYaw -= input.Delta.X * sens
	camPitch -= input.Delta.Y * sens
end)

UserInputService.InputChanged:Connect(function(input, processed)
	if processed or not camActive or not settings.wheelZoom then return end
	if input.UserInputType == Enum.UserInputType.MouseWheel then
		controls.cameraDistance.set(settings.cameraDistance - input.Position.Z * 2)
	end
end)

local function hideMarker()
	marker.Visible = false
end

local function step(dt)
	camera = workspace.CurrentCamera
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum or hum.Health <= 0 then
		currentTarget = nil
		activeAimPos = nil
		setTargetInfo(nil)
		setCircleLocked(false)
		hideMarker()
		exitCamera()
		return
	end

	-- FOV
	if savedFOV then
		local goalFov = settings.changeFOV and settings.fov or savedFOV
		camera.FieldOfView += (goalFov - camera.FieldOfView) * math.min(1, dt * 10)
	end

	local head = char:FindFirstChild("Head") or hrp
	local origin = head.Position
	local target, info = findTarget(origin, camera.CFrame.LookVector, char)
	currentTarget = target
	setCircleLocked(target ~= nil)

	local aimPos
	if target then
		setTargetInfo(target.DisplayName, info.dist, info.hum.Health, info.hum.MaxHealth, target.UserId)
		aimPos = info.part.Position + Vector3.new(0, settings.heightOffset, 0)
	else
		setTargetInfo(nil)
	end
	activeAimPos = aimPos

	-- Hedef işareti
	if aimPos and settings.showMarker then
		local sp = camera:WorldToViewportPoint(aimPos)
		if sp.Z > 0 then
			marker.Position = UDim2.fromOffset(sp.X, sp.Y)
			markerStroke.Color = THEME.green
			marker.Visible = true
		else
			marker.Visible = false
		end
	else
		marker.Visible = false
	end

	-- KİLİTLENMEMİŞ DAİRE MODU:
	-- Kamera ve karakter tamamen serbest kalır; hedefe kilitlenilse bile bakış açısı zorlanmaz,
	-- ancak activeAimPos üzerinden atılan skiller hedefi bulur.
	if settings.unlockedCircle then
		exitCamera()
		if hum.AutoRotate == false then hum.AutoRotate = true end
		return
	end

	-- KARAKTER MODU
	if settings.mode == "Karakter" then
		exitCamera()
		if not aimPos then
			if hum.AutoRotate == false then hum.AutoRotate = true end
			return
		end
		hum.AutoRotate = false
		local pos = hrp.Position
		local dir = aimPos - pos
		if Vector3.new(dir.X, 0, dir.Z).Magnitude > 0.05 then
			local look = hrp.CFrame.LookVector
			local curYaw = math.atan2(-look.X, -look.Z)
			local goalYaw = math.atan2(-dir.X, -dir.Z)
			hrp.CFrame = CFrame.new(pos) * CFrame.Angles(0, approach(curYaw, goalYaw, dt), 0)
		end
		return
	end

	-- KAMERA MODU
	if hum.AutoRotate == false then hum.AutoRotate = true end
	enterCamera()
	if camera.CameraType ~= Enum.CameraType.Scriptable then
		camera.CameraType = Enum.CameraType.Scriptable
	end

	local rmb = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
	local altHeld = UserInputService:IsKeyDown(Enum.KeyCode.LeftAlt)
	local autoLock = settings.circleEnabled and settings.circleMouseLock
		and not altHeld and not mouseOverMenu()
	local freeLook = (settings.lockMouse and not altHeld) or autoLock

	if freeLook then
		UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
	elseif rmb then
		UserInputService.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
	else
		UserInputService.MouseBehavior = Enum.MouseBehavior.Default
	end

	local delta = UserInputService:GetMouseDelta()
	local limit = math.rad(settings.pitchLimit)
	if freeLook or rmb then
		local sens = 0.003 * settings.sensitivity
		camYaw -= delta.X * sens
		camPitch -= delta.Y * sens
	end

	if aimPos then
		local dir = aimPos - origin
		if dir.Magnitude > 0.05 then
			camYaw = approach(camYaw, math.atan2(-dir.X, -dir.Z), dt)
			if not settings.cameraYawOnly then
				camPitch = approach(camPitch, math.asin(math.clamp(dir.Unit.Y, -1, 1)), dt)
			end
		end
	end
	camPitch = math.clamp(camPitch, -limit, limit)

	local rot = CFrame.Angles(0, camYaw, 0) * CFrame.Angles(camPitch, 0, 0)
	local pos = origin
	if settings.cameraDistance > 0.5 then
		local back = -rot.LookVector * settings.cameraDistance
		rayParams.FilterDescendantsInstances = { char }
		local hit = workspace:Raycast(origin, back, rayParams)
		if hit then
			pos = hit.Position + rot.LookVector * 0.5
		else
			pos = origin + back
		end
	end
	camera.CFrame = CFrame.new(pos) * rot
end

setRunning = function(state)
	if state == running then return end
	running = state
	if running then
		savedFOV = camera.FieldOfView
		runBtn.Text = "■  DURDUR"
		tween(runBtn, { BackgroundColor3 = THEME.red })
		mobileRunBtn.Text = "DURDUR"
		tween(mobileRunBtn, { BackgroundColor3 = THEME.red })
		statusText.Text = "Durum: Çalışıyor"
		statusText.TextColor3 = THEME.green
		pillText.Text = "AKTİF"
		pillText.TextColor3 = THEME.green
		tween(pill, { BackgroundColor3 = Color3.fromRGB(28, 70, 50) })
		statusDot.BackgroundColor3 = THEME.green
		dotStroke.Transparency = 0.2
		pulseTween = TweenService:Create(
			dotStroke,
			TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
			{ Transparency = 1 }
		)
		pulseTween:Play()
		RunService:BindToRenderStep(BIND_NAME, Enum.RenderPriority.Camera.Value + 1, step)
		if settings.silentUI and main.Visible then
			setMenuVisible(false)
			hiddenBySilent = true
		end
		notify("Başlatıldı")
	else
		pcall(function()
			RunService:UnbindFromRenderStep(BIND_NAME)
		end)
		exitCamera()
		local char = player.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum then hum.AutoRotate = true end
		if savedFOV then
			camera.FieldOfView = savedFOV
			savedFOV = nil
		end
		currentTarget = nil
		activeAimPos = nil
		setTargetInfo(nil)
		hideMarker()
		runBtn.Text = "▶  ÇALIŞTIR"
		tween(runBtn, { BackgroundColor3 = THEME.green })
		mobileRunBtn.Text = "BAŞLAT"
		tween(mobileRunBtn, { BackgroundColor3 = THEME.green })
		statusText.Text = "Durum: Kapalı"
		statusText.TextColor3 = THEME.subtext
		pillText.Text = "KAPALI"
		pillText.TextColor3 = THEME.subtext
		tween(pill, { BackgroundColor3 = THEME.off })
		statusDot.BackgroundColor3 = THEME.subtext
		if pulseTween then
			pulseTween:Cancel()
			pulseTween = nil
		end
		dotStroke.Transparency = 1
		if hiddenBySilent then
			setMenuVisible(true)
			hiddenBySilent = false
		end
		notify("Durduruldu")
	end
	circleLocked = false
	renderCircle()
	refreshDot()
end

UserInputService.InputBegan:Connect(function(input, processed)
	if listener and input.UserInputType == Enum.UserInputType.Keyboard then
		local l = listener
		listener = nil
		l(input.KeyCode)
		return
	end
	if processed then return end
	if input.KeyCode == settings.keyMenu then
		setMenuVisible(not main.Visible)
		hiddenBySilent = false
	elseif input.KeyCode == settings.keyCircle then
		controls.circleEnabled.set(not settings.circleEnabled)
		notify(settings.circleEnabled and "Daire: Açık" or "Daire: Kapalı")
	elseif input.KeyCode == settings.keyToggle then
		if settings.holdMode then
			setRunning(true)
		else
			setRunning(not running)
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if settings.holdMode and input.KeyCode == settings.keyToggle then
		setRunning(false)
	end
end)

	setRunning(false)
